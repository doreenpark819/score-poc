function stripBom(text) { return text.charCodeAt(0) === 0xFEFF ? text.slice(1) : text; }

// 바이트 → 텍스트. UTF-8로 읽히지 않으면 EUC-KR(CP949)로 다시 읽는다 (사내 시스템 CSV는 EUC-KR인 경우가 많음)
function decodeBytes(buf) {
  const bytes = buf instanceof Uint8Array ? buf : new Uint8Array(buf);
  try { return { text: stripBom(new TextDecoder('utf-8', { fatal: true }).decode(bytes)), encoding: 'UTF-8' }; }
  catch { return { text: stripBom(new TextDecoder('euc-kr').decode(bytes)), encoding: 'EUC-KR' }; }
}

// 첫 줄에서 가장 많이 나온 구분자 (따옴표 밖 기준): 쉼표, 탭, 세미콜론, 파이프
function detectDelimiter(text) {
  let end = text.indexOf('\n'); if (end < 0) end = text.length;
  const line = text.slice(0, end), cnt = { ',': 0, '\t': 0, ';': 0, '|': 0 };
  let q = false;
  for (const ch of line) { if (ch === '"') q = !q; else if (!q && ch in cnt) cnt[ch]++; }
  const [best, n] = Object.entries(cnt).sort((a, b) => b[1] - a[1])[0];
  return n > 0 ? best : ',';
}

// 컬럼명 표준화: 공백·밑줄·괄호를 무시하고 CONFIG.COLUMN_ALIASES의 별칭(한글 등)을 표준 이름으로
const normKey = h => String(h).toLowerCase().replace(/^\uFEFF/, '').replace(/[\s_\-()\[\]."']/g, '');
let ALIAS_INDEX = null;
function canonicalColumn(h) {
  if (!ALIAS_INDEX) {
    ALIAS_INDEX = new Map();
    for (const [canon, aliases] of Object.entries(CONFIG.COLUMN_ALIASES)) for (const a of [canon, ...aliases]) ALIAS_INDEX.set(normKey(a), canon);
  }
  return ALIAS_INDEX.get(normKey(h)) || String(h).trim();
}

// 따옴표가 있는 줄만 RFC4180 규칙으로 처리 (따옴표 안 구분자, "" 이스케이프)
function splitLine(line, delim = ',') {
  if (line.indexOf('"') < 0) return line.split(delim);
  const out = []; let f = '', q = false;
  for (let i = 0; i < line.length; i++) {
    const ch = line[i];
    if (q) { if (ch === '"') { if (line[i + 1] === '"') { f += '"'; i++; } else q = false; } else f += ch; }
    else if (ch === '"') q = true;
    else if (ch === delim) { out.push(f); f = ''; }
    else f += ch;
  }
  out.push(f);
  return out;
}

function readHeader(text, delim = detectDelimiter(stripBom(text))) {
  text = stripBom(text);
  let end = text.indexOf('\n'); if (end < 0) end = text.length;
  return splitLine(text.slice(0, end).replace(/\r$/, ''), delim).map(h => h.trim());
}

// 숫자 읽기: 엄격하게 먼저, 안 되면 천 단위 쉼표·공백·'원'·'₩'·'%'를 지우고 다시 (고친 건수는 따로 셈)
function toNumLoose(s) {
  const t = String(s).replace(/[,\s원₩%]/g, '');
  if (t === '' || !/^[-+]?(\d+\.?\d*|\.\d+)(e[-+]?\d+)?$/i.test(t)) return null;
  return Number(t);
}

// 열(column) 단위 파싱. keep에 든 컬럼만 보관하고, num 컬럼은 숫자(빈 값·오류는 null)로,
// 나머지는 같은 문자열을 공유(intern)해 메모리를 줄인다. 따옴표 안 줄바꿈은 지원하지 않음(입력 CSV에 없음)
function parseCSV(text, { keep = null, num = [], delim = null, rename = null } = {}) {
  text = stripBom(text);
  delim = delim || detectDelimiter(text);
  const rawHeader = readHeader(text, delim);
  const header = rename ? rawHeader.map(rename) : rawHeader;
  const cols = {}, tIdx = [], tArr = [], tNum = []; // 보관할 열 번호·저장 배열·숫자 여부
  const numFixed = {}, numInvalid = {};           // 숫자 열: 자동 수정 / 숫자 아님 (건수, 예시)
  header.forEach((h, i) => {
    if (keep && !keep.includes(h)) return;
    cols[h] = [];
    tIdx.push(i); tArr.push(cols[h]); tNum.push(num.includes(h));
  });
  const nT = tIdx.length;
  const pool = new Map();
  const intern = s => { let v = pool.get(s); if (v === undefined) { pool.set(s, s); v = s; } return v; };
  let pos = text.indexOf('\n'); pos = pos < 0 ? text.length : pos + 1;
  let length = 0;
  while (pos < text.length) {
    let end = text.indexOf('\n', pos); if (end < 0) end = text.length;
    let line = text.slice(pos, end); pos = end + 1;
    if (line.charCodeAt(line.length - 1) === 13) line = line.slice(0, -1);
    if (line.length === 0 || (line.charCodeAt(0) <= 32 && line.trim() === '')) continue;
    const v = splitLine(line, delim);
    for (let t = 0; t < nT; t++) {
      let s = v[tIdx[t]];
      if (s === undefined) s = '';
      else if (s.length && (s.charCodeAt(0) <= 32 || s.charCodeAt(s.length - 1) <= 32)) s = s.trim();
      if (!tNum[t]) { tArr[t].push(intern(s)); continue; }
      let n = toNum(s);
      if (n === null && s !== '') {
        const h = header[tIdx[t]];
        n = toNumLoose(s);
        const bag = n === null ? numInvalid : numFixed;
        const e = bag[h] || (bag[h] = { count: 0, examples: [] });
        e.count++; if (e.examples.length < 3) e.examples.push(`${length + 2}행: "${s}"`);
      }
      tArr[t].push(n);
    }
    length++;
  }
  return { header, rawHeader, delim, length, cols, numFixed, numInvalid };
}

// 헤더로 파일 종류 판별: 필수 컬럼을 모두 가진 spec 중 컬럼 수가 가장 많은(가장 구체적인) 것
function detectFileType(header) {
  const set = new Set(header);
  const hits = FILE_SPECS.filter(s => s.cols.every(c => set.has(c)));
  hits.sort((a, b) => b.cols.length - a.cols.length);
  return hits.length ? hits[0].key : null;
}

// 판별 + 필요한 컬럼만 파싱 (구분자 자동 판별, 컬럼명 표준화)
function parseFile(text, { encoding = null } = {}) {
  text = stripBom(text);
  const delim = detectDelimiter(text);
  const rawHeader = readHeader(text, delim);
  const header = rawHeader.map(canonicalColumn);
  const key = detectFileType(header);
  if (!key) return { key: null, header, rawHeader, delim, encoding };
  const spec = FILE_SPECS.find(s => s.key === key);
  const renamed = rawHeader.map((h, i) => [h, header[i]]).filter(([a, b]) => a !== b && [...spec.cols, ...spec.extra].includes(b));
  return { key, encoding, renamed, ...parseCSV(text, { keep: [...spec.cols, ...spec.extra], num: spec.num, delim, rename: canonicalColumn }) };
}

function toNum(v) {
  const n = Number(v);
  return v === '' || !Number.isFinite(n) ? null : n;
}

// 키 컬럼 값 → 행 번호
function indexCol(table, key) {
  const m = new Map(), col = table.cols[key];
  for (let i = 0; i < table.length; i++) m.set(col[i], i);
  return m;
}

// files: { contracts, coverages?, loss, credit, delinq?, apps, appCoverages } 각각 parseFile 결과
// 반환
//  policies: 증권 단위(담보 실적 합산). PRD 4장 계산은 모두 이 단위
//  cov: 증권×담보 단위 열 배열 (세그먼트 분석용). pol = policies 인덱스
function buildDataset(files) {
  const creditIx = indexCol(files.credit, 'customer_id');
  const delinqIx = files.delinq ? indexCol(files.delinq, 'customer_id') : null;
  const hasDelinq = !!delinqIx;
  const C = files.contracts.cols, CR = files.credit.cols, DQ = files.delinq?.cols;
  const delinqOf = j => {
    const o = {};
    for (const k of [...CONFIG.DELINQ.DEFAULT_COLS, ...CONFIG.DELINQ.DELINQ_COLS]) o[k] = DQ[k][j];
    return o;
  };

  const counts = { contracts: files.contracts.length, withLoss: 0, withCredit: 0, withDelinq: 0, badNumber: 0, analyzed: 0,
    lossRows: files.loss.length, lossRowsUnmatched: 0, lossRowsBad: 0, lossRowsNoSI: 0, coverageRowsUsed: 0,
    apps: files.apps.length, appsWithCredit: 0, appsWithDelinq: 0, appsWithCoverage: 0, appCoverageRows: files.appCoverages.length };

  // 1) 담보 실적 → 증권별 합산
  const L = files.loss.cols;
  const lossAgg = new Map(); // policy_no → [premium, loss, 담보행 시작 목록]
  const badPolicy = new Set();
  for (let i = 0; i < files.loss.length; i++) {
    const p = L.risk_premium[i], paid = L.claim_paid[i], res = L.reserve_change[i];
    if (p === null || paid === null || res === null) { counts.lossRowsBad++; badPolicy.add(L.policy_no[i]); continue; }
    let a = lossAgg.get(L.policy_no[i]);
    if (!a) { a = { premium: 0, loss: 0, rows: [] }; lossAgg.set(L.policy_no[i], a); }
    a.premium += p; a.loss += paid + res; a.rows.push(i);
  }

  // 2) 계약 + 신용 + 연체 조인
  const policies = [];
  for (let i = 0; i < files.contracts.length; i++) {
    const pno = C.policy_no[i], agg = lossAgg.get(pno);
    const ci = creditIx.get(C.customer_id[i]);
    const di = delinqIx ? delinqIx.get(C.customer_id[i]) : undefined;
    if (agg) counts.withLoss++;
    if (ci !== undefined) counts.withCredit++;
    if (di !== undefined) counts.withDelinq++;
    if (!agg || ci === undefined) continue;
    // 버전 B를 계산할 때는 연체정보도 있어야 분석 대상 (A·B를 같은 계약 집합으로 비교)
    if (hasDelinq && di === undefined) continue;
    const decile = CR.credit_decile[ci];
    if (decile === null || badPolicy.has(pno)) { counts.badNumber++; continue; }
    policies.push({ policy_no: pno, customer_id: C.customer_id[i], decile, premium: agg.premium, loss: agg.loss,
      delinq: hasDelinq ? delinqOf(di) : null, grade: hasDelinq ? delinqGrade(delinqOf(di)) : null,
      age: C.age ? C.age[i] : null, gender: C.gender ? C.gender[i] || null : null, _rows: agg.rows });
  }
  counts.analyzed = policies.length;
  const used = new Set(policies.map(p => p.policy_no));
  for (const pno of lossAgg.keys()) if (!used.has(pno)) counts.lossRowsUnmatched += lossAgg.get(pno).rows.length;

  // 3) 담보 단위 열 배열 (가입금액은 contract_coverages에서)
  const siIx = new Map(); // policy_no → [[coverage_code, 행 번호], ...] (증권당 담보 14개 이하)
  if (files.coverages) {
    const V = files.coverages.cols;
    for (let i = 0; i < files.coverages.length; i++) {
      let list = siIx.get(V.policy_no[i]);
      if (!list) { list = []; siIx.set(V.policy_no[i], list); }
      list.push([V.coverage_code[i], i]);
    }
  }
  const siRow = (pno, code) => { const hit = siIx.get(pno)?.find(e => e[0] === code); return hit ? hit[1] : undefined; };
  const nCov = policies.reduce((s, p) => s + p._rows.length, 0);
  const coverageNames = [], coverageCodes = [], coverageIdx = new Map();
  const cov = { length: nCov, pol: new Int32Array(nCov), coverage: new Uint16Array(nCov),
    sumInsured: new Float64Array(nCov).fill(NaN), premium: new Float64Array(nCov), loss: new Float64Array(nCov),
    claims: new Float64Array(nCov), names: coverageNames, codes: coverageCodes };
  const hasClaimCnt = !!L.claim_cnt;
  let k = 0;
  policies.forEach((p, pi) => {
    for (const r of p._rows) {
      const code = L.coverage_code[r];
      const vi = siRow(p.policy_no, code);
      const name = vi !== undefined ? files.coverages.cols.coverage_name[vi] : code;
      let ci = coverageIdx.get(name);
      if (ci === undefined) { ci = coverageNames.length; coverageNames.push(name); coverageCodes.push(code); coverageIdx.set(name, ci); }
      cov.pol[k] = pi; cov.coverage[k] = ci;
      if (vi !== undefined && files.coverages.cols.sum_insured[vi] !== null) cov.sumInsured[k] = files.coverages.cols.sum_insured[vi];
      else counts.lossRowsNoSI++;
      cov.premium[k] = L.risk_premium[r]; cov.loss[k] = L.claim_paid[r] + L.reserve_change[r];
      if (hasClaimCnt) cov.claims[k] = L.claim_cnt[r] || 0;
      k++;
    }
    delete p._rows;
  });
  counts.coverageRowsUsed = nCov;

  // 4) 청약 + 담보
  const A = files.apps.cols, AC = files.appCoverages.cols;
  const appCov = new Map();
  for (let i = 0; i < files.appCoverages.length; i++) {
    let list = appCov.get(AC.app_no[i]);
    if (!list) { list = []; appCov.set(AC.app_no[i], list); }
    list.push({ code: AC.coverage_code[i], name: AC.coverage_name[i], sum_insured: AC.sum_insured[i] });
  }
  const apps = [];
  for (let i = 0; i < files.apps.length; i++) {
    const ci = creditIx.get(A.customer_id[i]);
    const di = delinqIx ? delinqIx.get(A.customer_id[i]) : undefined;
    const coverages = appCov.get(A.app_no[i]) || [];
    if (ci !== undefined) counts.appsWithCredit++;
    if (di !== undefined) counts.appsWithDelinq++;
    if (coverages.length) counts.appsWithCoverage++;
    apps.push({ app_no: A.app_no[i], customer_id: A.customer_id[i],
      age: A.age ? A.age[i] : null, gender: A.gender ? A.gender[i] || null : null,
      decile: ci !== undefined ? CR.credit_decile[ci] : null,
      delinq: di !== undefined ? delinqOf(di) : null, grade: di !== undefined ? delinqGrade(delinqOf(di)) : null, coverages });
  }

  return { hasDelinq, hasClaimCnt, counts, policies, cov, apps };
}

// 손해율 = (Σ보험금 + Σ준비금변동) ÷ Σ위험보험료
function lossRatio(policies) {
  let p = 0, l = 0;
  for (const x of policies) { p += x.premium; l += x.loss; }
  return p > 0 ? l / p : null;
}
// ② 10분위 상대도 = 분위 손해율 ÷ 전체 손해율
// opts.monotone: true면 PAVA 단조 보정한 상대도를 rel에 넣음 (원래 값은 rawRel). 이후 계산은 모두 rel 사용
function decileTable(policies, { monotone = CONFIG.MONOTONE_DECILE } = {}) {
  const overall = lossRatio(policies);
  const prem = new Array(CONFIG.DECILES + 1).fill(0), loss = prem.slice(), n = prem.slice();
  for (const p of policies) { prem[p.decile] += p.premium; loss[p.decile] += p.loss; n[p.decile]++; }
  const rows = [];
  for (let d = 1; d <= CONFIG.DECILES; d++) {
    const lr = prem[d] > 0 ? loss[d] / prem[d] : null;
    const rel = lr != null && overall ? lr / overall : null;
    rows.push({ decile: d, n: n[d], premium: prem[d], loss: loss[d], lr, rawRel: rel, rel, block: String(d) });
  }
  if (monotone) {
    for (const b of monotoneBlocks(rows.filter(r => r.premium > 0))) {
      const label = b.rows.length > 1 ? `${b.rows[0].decile}~${b.rows[b.rows.length - 1].decile}` : String(b.rows[0].decile);
      for (const r of b.rows) { r.rel = (b.loss / b.premium) / overall; r.block = label; }
    }
  }
  return { overall, rows, monotone };
}

// PAVA(Pool Adjacent Violators): 손해율이 앞 묶음보다 낮아지면 합쳐서 묶음 손해율 = Σ실제손해 ÷ Σ위험보험료
// 위험보험료 가중이므로 전체 Σ(위험보험료 × 묶음 손해율) = Σ실제손해가 보존됨
function monotoneBlocks(rows) {
  const blocks = [];
  for (const r of rows) {
    blocks.push({ rows: [r], premium: r.premium, loss: r.loss });
    while (blocks.length > 1) {
      const b = blocks[blocks.length - 1], a = blocks[blocks.length - 2];
      if (a.loss / a.premium <= b.loss / b.premium) break;
      blocks.splice(-2, 2, { rows: [...a.rows, ...b.rows], premium: a.premium + b.premium, loss: a.loss + b.loss });
    }
  }
  return blocks;
}
// ① 연체등급: 2 채무불이행 / 1 연체 / 0 정상 (연체정보 없으면 null)
function delinqGrade(d) {
  if (!d) return null;
  const sum = cols => cols.reduce((s, c) => s + (d[c] || 0), 0);
  if (sum(CONFIG.DELINQ.DEFAULT_COLS) >= 1) return 2;
  if (sum(CONFIG.DELINQ.DELINQ_COLS) >= 1) return 1;
  return 0;
}

// ③ 연체등급별 단순·보정 상대도
//  단순 = 등급 손해율 ÷ 전체 손해율
//  보정 = 등급 Σ실제손해 ÷ 등급 Σ(위험보험료 × 전체손해율 × 10분위 상대도)  (신용평점과의 이중반영 제거)
function delinqTable(policies, dec) {
  const rows = CONFIG.DELINQ.LABELS.map((label, g) => ({ grade: g, label, n: 0, premium: 0, loss: 0, expected: 0 }));
  for (const p of policies) {
    if (p.grade == null) continue;
    const r = rows[p.grade];
    r.n++; r.premium += p.premium; r.loss += p.loss;
    r.expected += p.premium * dec.overall * dec.rows[p.decile - 1].rel;
  }
  for (const r of rows) {
    r.lr = r.premium > 0 ? r.loss / r.premium : null;
    r.simpleRel = r.lr != null ? r.lr / dec.overall : null;
    r.adjRel = r.expected > 0 ? r.loss / r.expected : null;
  }
  return rows;
}

// ④ 종합상대도: A = 10분위 상대도, B = 10분위 상대도 × 연체 보정 상대도
function makeRelFns(dec, dq) {
  const decRel = d => dec.rows[d - 1].rel;
  return {
    A: x => decRel(x.decile),
    B: dq ? x => decRel(x.decile) * dq[x.grade].adjRel : null,
  };
}
const score = rel => Math.round(100 / rel);

// ⑥ 분리도: 종합상대도 오름차순(동점은 policy_no 순)으로 N등분 → 최악 구간 손해율 ÷ 최우량 구간 손해율
function separation(policies, relFn, buckets = CONFIG.SEPARATION_BUCKETS) {
  const order = policies.map(p => ({ p, r: relFn(p) }))
    .sort((a, b) => a.r - b.r || (a.p.policy_no < b.p.policy_no ? -1 : a.p.policy_no > b.p.policy_no ? 1 : 0));
  const n = order.length, groups = [];
  for (let i = 0; i < buckets; i++) {
    const part = order.slice(Math.floor(i * n / buckets), Math.floor((i + 1) * n / buckets)).map(o => o.p);
    groups.push({ bucket: i + 1, n: part.length, lr: lossRatio(part),
      relMin: order[Math.floor(i * n / buckets)].r, relMax: order[Math.floor((i + 1) * n / buckets) - 1].r });
  }
  return { ratio: groups[buckets - 1].lr / groups[0].lr, groups };
}

// ─ 세그먼트 × 연체등급 교차표 (담보 단위) ─
// dim: 'coverage' | 'gender' | 'age' | 'sumInsured'
// coverage: 특정 담보만 볼 때 담보 인덱스 (sumInsured는 필수)
// 칸마다 담보행 수·사고 건수·손해율·보정 A/E(= Σ실제손해 ÷ Σ위험보험료×전체손해율×10분위 상대도)
// 정상 대비 배수 = 해당 등급 보정 A/E ÷ 정상 보정 A/E (같은 세그먼트 안에서 비교 → 담보·연령 수준 차이 제거)
function ageBand(age) {
  if (age == null) return null;
  const b = CONFIG.SEGMENT.AGE_BANDS.find(b => age >= b.min && age <= b.max);
  return b ? b.label : null;
}
function segmentTable(ds, dec, { dim, coverage = null } = {}) {
  const { cov, policies } = ds;
  const keyOf = {
    coverage: k => ds.cov.names[cov.coverage[k]],
    gender: k => ({ M: '남', F: '여' }[policies[cov.pol[k]].gender] || null),
    age: k => ageBand(policies[cov.pol[k]].age),
    sumInsured: k => Number.isNaN(cov.sumInsured[k]) ? null : cov.sumInsured[k],
  }[dim];
  const decRel = dec.rows.map(r => r.rel);
  const map = new Map();
  for (let k = 0; k < cov.length; k++) {
    if (coverage != null && cov.coverage[k] !== coverage) continue;
    const p = policies[cov.pol[k]];
    if (p.grade == null) continue;
    const key = keyOf(k);
    if (key == null) continue;
    let seg = map.get(key);
    if (!seg) {
      seg = { key, cells: [0, 1, 2].map(() => ({ n: 0, claims: 0, premium: 0, loss: 0, expected: 0 })) };
      map.set(key, seg);
    }
    const c = seg.cells[p.grade];
    c.n++; c.claims += cov.claims[k]; c.premium += cov.premium[k]; c.loss += cov.loss[k];
    c.expected += cov.premium[k] * dec.overall * decRel[p.decile - 1];
  }
  const order = {
    coverage: () => ds.cov.names.map((n, i) => [ds.cov.codes[i], n]).sort((a, b) => a[0] < b[0] ? -1 : 1).map(e => e[1]),
    gender: () => ['남', '여'],
    age: () => CONFIG.SEGMENT.AGE_BANDS.map(b => b.label),
    sumInsured: () => [...map.keys()].sort((a, b) => a - b),
  }[dim]();
  const segs = order.filter(k => map.has(k)).map(k => map.get(k));
  for (const s of segs) {
    for (const c of s.cells) {
      c.lr = c.premium > 0 ? c.loss / c.premium : null;
      c.ae = c.expected > 0 ? c.loss / c.expected : null;
      c.credible = !ds.hasClaimCnt || c.claims >= CONFIG.SEGMENT.MIN_CLAIMS;
    }
    const base = s.cells[0].ae;
    for (const c of s.cells) c.multiple = base > 0 && c.ae != null ? c.ae / base : null;
    const tot = s.cells.reduce((a, c) => ({ n: a.n + c.n, claims: a.claims + c.claims, premium: a.premium + c.premium, loss: a.loss + c.loss }),
      { n: 0, claims: 0, premium: 0, loss: 0 });
    s.total = { ...tot, lr: tot.premium > 0 ? tot.loss / tot.premium : null };
  }
  return segs;
}
// ⑤ 권고 인수조건: 종합상대도가 구간 상한(max) 미만인 첫 규칙
function uwRule(rel) { return CONFIG.UW_RULES.find(r => rel < r.max); }

// 청약 심사: 상대도 내역·스코어·권고(B, 연체정보 없으면 A)·담보별 권고 가입금액
function assessApps(ds, dec, dq) {
  const excl = new Set(CONFIG.EXCLUSION_COVERAGES);
  return ds.apps.map(a => {
    const out = { ...a, relDecile: null, relDelinq: null, relA: null, relB: null, scoreA: null, scoreB: null,
      basis: null, rule: null, status: null, coverageResults: [] };
    if (a.decile == null || !dec.rows[a.decile - 1] || dec.rows[a.decile - 1].rel == null) {
      out.status = '신용정보 없음';
      out.coverageResults = a.coverages.map(c => ({ ...c, pct: null, recommended: null, note: '평가 불가' }));
      return out;
    }
    out.relDecile = dec.rows[a.decile - 1].rel;
    out.relA = out.relDecile; out.scoreA = score(out.relA);
    if (dq && a.grade != null) {
      out.relDelinq = dq[a.grade].adjRel;
      out.relB = out.relDecile * out.relDelinq; out.scoreB = score(out.relB);
    }
    out.basis = out.relB != null ? 'B' : 'A';
    out.rule = uwRule(out.basis === 'B' ? out.relB : out.relA);
    out.status = out.rule.label;
    out.coverageResults = a.coverages.map(c => {
      if (out.rule.limitPct == null) return { ...c, pct: null, recommended: null, note: '심사역 판단' };
      if (out.rule.exclusion && excl.has(c.name)) return { ...c, pct: 0, recommended: 0, note: '부담보' };
      return { ...c, pct: out.rule.limitPct, recommended: c.sum_insured == null ? null : Math.round(c.sum_insured * out.rule.limitPct / 100),
        note: out.rule.limitPct < 100 ? '한도축소' : '' };
    });
    return out;
  });
}

function uwSummary(assessed) {
  const m = new Map(CONFIG.UW_RULES.map(r => [r.label, 0]));
  for (const a of assessed) m.set(a.status, (m.get(a.status) || 0) + 1);
  return m;
}

// 결과 CSV (청약 × 담보 한 행). 엑셀 한글 깨짐 방지를 위해 BOM 포함
function csvCell(v) {
  if (v == null) return '';
  const s = typeof v === 'number' ? (Number.isInteger(v) ? String(v) : v.toFixed(4)) : String(v);
  return /[",\r\n]/.test(s) ? '"' + s.replace(/"/g, '""') + '"' : s;
}
function exportCsv(assessed) {
  const header = ['app_no', 'customer_id', 'age', 'gender', 'credit_decile', 'delinq_grade', 'delinq_label',
    'rel_decile', 'rel_delinq_adj', 'rel_A', 'rel_B', 'score_A', 'score_B', 'basis', 'uw_recommendation',
    'coverage_code', 'coverage_name', 'sum_insured', 'recommended_pct', 'recommended_sum_insured', 'note'];
  const lines = [header.join(',')];
  for (const a of assessed) {
    const base = [a.app_no, a.customer_id, a.age, a.gender, a.decile, a.grade,
      a.grade == null ? null : CONFIG.DELINQ.LABELS[a.grade], a.relDecile, a.relDelinq, a.relA, a.relB,
      a.scoreA, a.scoreB, a.basis, a.status];
    const covs = a.coverageResults.length ? a.coverageResults : [{}];
    for (const c of covs) lines.push([...base, c.code, c.name, c.sum_insured, c.pct, c.recommended, c.note].map(csvCell).join(','));
  }
  return '﻿' + lines.join('\r\n') + '\r\n';
}
// ─ 검증·보정 (실데이터 대비) ─────────────────────────────────────────

// 시드 고정 난수 (붓스트랩·교차 검증이 매번 같은 결과를 내도록)
function seededRandom(seed) {
  let s = seed | 0;
  return () => {
    s = (s + 0x6D2B79F5) | 0;
    let t = Math.imul(s ^ (s >>> 15), 1 | s);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

// 손해 기준: 증권×담보 행의 손해를 무엇으로 볼지 정한다. 세 기준 모두 전체 손해 합계를 보존한다.
//  raw       원손해 (보험금 + 준비금변동)
//  capped    대형사고 상한: 사고 행 손해의 상위 (1−CAP.QUANTILE) 금액에서 자르고, 잘린 초과분은 전체에 비례 배분
//  frequency 빈도 기준: 사고 건수 × 담보별 평균 1건당 손해 (금액 우연성을 제거하고 '사고가 났는가'만 반영)
// 반환: 손해만 바꾼 데이터셋 사본 (policies[].loss, cov.loss 교체) + basisInfo
function applyLossBasis(ds, basis = CONFIG.LOSS_BASIS) {
  const cov = ds.cov, n = cov.length;
  if (basis === 'frequency' && !ds.hasClaimCnt) basis = 'raw';
  if (basis === 'raw') return { ...ds, basis, basisInfo: { label: LOSS_BASIS_LABELS.raw } };
  const adj = new Float64Array(n);
  let total = 0, info;
  for (let k = 0; k < n; k++) total += cov.loss[k];
  if (basis === 'capped') {
    const pos = [];
    for (let k = 0; k < n; k++) if (ds.hasClaimCnt ? cov.claims[k] > 0 : cov.loss[k] > 0) pos.push(cov.loss[k]);
    pos.sort((a, b) => a - b);
    const threshold = pos.length ? pos[Math.floor(CONFIG.CAP.QUANTILE * (pos.length - 1))] : Infinity;
    let capped = 0, rows = 0;
    for (let k = 0; k < n; k++) { adj[k] = Math.min(cov.loss[k], threshold); capped += adj[k]; if (cov.loss[k] > threshold) rows++; }
    const factor = capped > 0 ? total / capped : 1;
    for (let k = 0; k < n; k++) adj[k] *= factor;
    info = { label: LOSS_BASIS_LABELS.capped, threshold, factor, cappedRows: rows, excessShare: total > 0 ? 1 - capped / total : 0 };
  } else {
    const m = cov.names.length, sumL = new Float64Array(m), sumC = new Float64Array(m);
    for (let k = 0; k < n; k++) { sumL[cov.coverage[k]] += cov.loss[k]; sumC[cov.coverage[k]] += cov.claims[k]; }
    const sev = Array.from(sumL, (l, c) => sumC[c] > 0 ? l / sumC[c] : null);
    for (let k = 0; k < n; k++) { const s = sev[cov.coverage[k]]; adj[k] = s == null ? cov.loss[k] : cov.claims[k] * s; }
    info = { label: LOSS_BASIS_LABELS.frequency, severity: Object.fromEntries(cov.names.map((nm, c) => [nm, sev[c]])) };
  }
  const pl = new Float64Array(ds.policies.length);
  for (let k = 0; k < n; k++) pl[cov.pol[k]] += adj[k];
  const policies = ds.policies.map((p, i) => ({ ...p, loss: pl[i] }));
  return { ...ds, policies, cov: { ...cov, loss: adj }, basis, basisInfo: info };
}
const LOSS_BASIS_LABELS = { raw: '원손해', capped: '대형사고 상한', frequency: '빈도 기준' };

// 칸(10분위 × 연체등급) 합계 → 지표. 앱의 상대도 공식은 모두 칸 합계의 함수라 붓스트랩을 빠르게 할 수 있다
function relFromCells(prem, loss, { monotone = false, hasGrade = true } = {}) {
  const D = CONFIG.DECILES, G = hasGrade ? 3 : 1;
  let P = 0, L = 0;
  const dp = new Array(D).fill(0), dl = dp.slice(), gp = new Array(G).fill(0), gl = gp.slice();
  for (let d = 0; d < D; d++) for (let g = 0; g < G; g++) {
    const k = d * G + g; P += prem[k]; L += loss[k]; dp[d] += prem[k]; dl[d] += loss[k]; gp[g] += prem[k]; gl[g] += loss[k];
  }
  const overall = P > 0 ? L / P : null;
  let decRel = dl.map((l, d) => dp[d] > 0 && overall ? (l / dp[d]) / overall : null);
  if (monotone) {
    const rows = dp.map((p, d) => ({ decile: d + 1, premium: p, loss: dl[d] })).filter(r => r.premium > 0);
    for (const b of monotoneBlocks(rows)) for (const r of b.rows) decRel[r.decile - 1] = (b.loss / b.premium) / overall;
  }
  if (!hasGrade) return { overall, decRel };
  const simpleRel = gl.map((l, g) => gp[g] > 0 && overall ? (l / gp[g]) / overall : null);
  const adjRel = [0, 1, 2].map(g => {
    let e = 0; for (let d = 0; d < D; d++) e += prem[d * G + g] * overall * (decRel[d] ?? 1);
    return e > 0 ? gl[g] / e : null;
  });
  return { overall, decRel, simpleRel, adjRel };
}

// 붓스트랩 신뢰구간: 증권을 복원추출해 상대도를 B번 다시 계산, 백분위 구간
function bootstrapCI(policies, { B = CONFIG.CI.BOOT, level = CONFIG.CI.LEVEL, seed = CONFIG.CI.SEED, monotone = false } = {}) {
  const hasGrade = policies.length > 0 && policies[0].grade != null;
  const G = hasGrade ? 3 : 1, C = CONFIG.DECILES * G, n = policies.length;
  const cell = new Int32Array(n), prem = new Float64Array(n), loss = new Float64Array(n);
  policies.forEach((p, i) => { cell[i] = (p.decile - 1) * G + (hasGrade ? p.grade : 0); prem[i] = p.premium; loss[i] = p.loss; });
  const rnd = seededRandom(seed), samples = [];
  for (let b = 0; b < B; b++) {
    const bp = new Float64Array(C), bl = new Float64Array(C);
    for (let j = 0; j < n; j++) { const i = (rnd() * n) | 0; bp[cell[i]] += prem[i]; bl[cell[i]] += loss[i]; }
    samples.push(relFromCells(bp, bl, { monotone, hasGrade }));
  }
  const a = (1 - level) / 2;
  const q = (vals, p) => { const v = vals.filter(x => x != null).sort((x, y) => x - y); if (!v.length) return null; const i = (v.length - 1) * p, lo = Math.floor(i); return v[lo] + (v[Math.ceil(i)] - v[lo]) * (i - lo); };
  const band = get => { const v = samples.map(get); const lo = q(v, a), hi = q(v, 1 - a); const m = v.filter(x => x != null); const mu = m.reduce((s, x) => s + x, 0) / (m.length || 1); const se = Math.sqrt(m.reduce((s, x) => s + (x - mu) ** 2, 0) / Math.max(1, m.length - 1)); return { lo, hi, se }; };
  return {
    B, level,
    overall: band(s => s.overall),
    decRel: Array.from({ length: CONFIG.DECILES }, (_, d) => band(s => s.decRel[d])),
    simpleRel: hasGrade ? [0, 1, 2].map(g => band(s => s.simpleRel[g])) : null,
    adjRel: hasGrade ? [0, 1, 2].map(g => band(s => s.adjRel[g])) : null,
  };
}

// 교차 검증 분리도: 증권을 무작위로 반으로 나눠 한쪽으로 상대도를 추정하고 다른 쪽에서 분리도를 잰다(양방향 평균).
// 같은 데이터로 정렬·채점하는 분리도는 과적합으로 부풀려지므로(통계 검증 VALIDATION.md) 이 값을 주 지표로 쓴다.
function separationCV(policies, { repeats = CONFIG.CV.REPEATS, seed = CONFIG.CV.SEED, monotone = false } = {}) {
  const rnd = seededRandom(seed), hasGrade = policies.length > 0 && policies[0].grade != null;
  const runs = [];
  for (let r = 0; r < repeats; r++) {
    const halves = [[], []];
    for (const p of policies) halves[rnd() < 0.5 ? 0 : 1].push(p);
    let A = 0, Bv = 0;
    for (const [tr, te] of [[0, 1], [1, 0]]) {
      const dec = decileTable(halves[tr], { monotone });
      const fill = rows => rows.map(x => x.rel ?? 1);
      const decRel = fill(dec.rows), relA = p => decRel[p.decile - 1];
      A += separation(halves[te], relA).ratio / 2;
      if (hasGrade) {
        const dq = delinqTable(halves[tr], dec), adj = dq.map(x => x.adjRel ?? 1);
        Bv += separation(halves[te], p => decRel[p.decile - 1] * adj[p.grade]).ratio / 2;
      }
    }
    runs.push({ A, B: hasGrade ? Bv : null });
  }
  const mean = k => runs.reduce((s, x) => s + x[k], 0) / runs.length;
  return { A: mean('A'), B: hasGrade ? mean('B') : null, BgtA: hasGrade ? runs.filter(x => x.B > x.A).length / runs.length : null, runs, repeats };
}
// ─ 데이터 자동 검토·전처리 ─────────────────────────────────────────
// 원본(parseFile 결과)은 그대로 두고 정제본을 새로 만든다. issues: 화면·리포트에 보일 검토 결과
//  level: error(행 제외·값 무효) | warn(확인 필요, 값은 유지) | fix(자동 수정) | info(참고)
const FILE_LABEL = Object.fromEntries(FILE_SPECS.map(s => [s.key, s.name]));
function reviewDataset(raw) {
  const issues = [], out = {};
  const add = (file, level, message, count = null, examples = [], action = '') =>
    issues.push({ file: file ? FILE_LABEL[file] || file : '전체', level, message, count, examples: examples.slice(0, 3), action });
  const copy = t => ({ ...t, cols: { ...t.cols } });
  const filterRows = (t, keep) => {
    const idx = []; for (let i = 0; i < t.length; i++) if (keep(i)) idx.push(i);
    const cols = {}; for (const c in t.cols) cols[c] = idx.map(i => t.cols[c][i]);
    return { ...t, cols, length: idx.length };
  };
  const ex = (t, rows, col) => rows.slice(0, 3).map(i => `${i + 2}행${col ? `: ${col}="${t.cols[col][i]}"` : ''}`);

  // 1) 읽기 단계 결과 (인코딩, 구분자, 컬럼명 매핑, 숫자 정리)
  for (const [key, t] of Object.entries(raw)) {
    if (t.encoding === 'EUC-KR') add(key, 'fix', 'UTF-8이 아니어서 EUC-KR(CP949)로 읽었습니다', null, [], '자동 판별');
    if (t.delim && t.delim !== ',') add(key, 'info', `구분자가 쉼표가 아니라 '${t.delim === '\t' ? '탭' : t.delim}'입니다`, null, [], '자동 판별');
    if (t.renamed?.length) add(key, 'fix', '컬럼명을 표준 이름으로 바꿨습니다', t.renamed.length, t.renamed.map(([a, b]) => `${a} → ${b}`), '자동 매핑');
    for (const [c, e] of Object.entries(t.numFixed || {})) add(key, 'fix', `${c}: 천 단위 쉼표·'원' 등 숫자 표기를 정리했습니다`, e.count, e.examples, '숫자로 변환');
    for (const [c, e] of Object.entries(t.numInvalid || {})) add(key, 'error', `${c}: 숫자가 아닌 값이 있습니다`, e.count, e.examples, '빈 값으로 처리');
  }

  // 2) 파일별 값 점검
  const normGender = v => { const s = String(v).trim().toLowerCase(); return ['m', '남', '남자', '1', 'male'].includes(s) ? 'M' : ['f', '여', '여자', '2', 'female'].includes(s) ? 'F' : s === '' ? '' : null; };
  const [aMin, aMax] = CONFIG.REVIEW.AGE_RANGE;
  const person = key => {
    let t = raw[key]; if (!t) return;
    t = copy(t);
    const id = key === 'contracts' ? 'policy_no' : 'app_no';
    const emptyRows = []; for (let i = 0; i < t.length; i++) if (!t.cols[id][i] || !t.cols.customer_id[i]) emptyRows.push(i);
    if (emptyRows.length) { add(key, 'error', `${id} 또는 customer_id가 비어 있는 행`, emptyRows.length, ex(t, emptyRows), '행 제외'); const bad = new Set(emptyRows); t = filterRows(t, i => !bad.has(i)); }
    const seen = new Set(), dups = []; for (let i = 0; i < t.length; i++) { const k = t.cols[id][i]; if (seen.has(k)) dups.push(i); else seen.add(k); }
    if (dups.length) { add(key, 'warn', `${id} 중복`, dups.length, ex(t, dups, id), '첫 행만 사용'); const bad = new Set(dups); t = filterRows(t, i => !bad.has(i)); }
    if (t.cols.gender) {
      let fixed = 0; const badRows = [];
      t.cols.gender = t.cols.gender.map((v, i) => { const g = normGender(v); if (g === null) { badRows.push(i); return ''; } if (g !== v) fixed++; return g; });
      if (fixed) add(key, 'fix', '성별 표기를 M/F로 통일했습니다 (남/여, 1/2 등)', fixed, [], '자동 변환');
      if (badRows.length) add(key, 'warn', '알 수 없는 성별 값', badRows.length, ex(raw[key], badRows, 'gender'), '세그먼트 분석에서 제외');
    }
    if (t.cols.age) {
      const badRows = []; t.cols.age = t.cols.age.map((v, i) => { if (v != null && (v < aMin || v > aMax)) { badRows.push(i); return null; } return v; });
      if (badRows.length) add(key, 'warn', `연령이 ${aMin}~${aMax} 범위를 벗어남`, badRows.length, badRows.slice(0, 3).map(i => `${i + 2}행`), '연령 비움');
    }
    out[key] = t;
  };
  person('contracts'); person('apps');

  for (const key of ['coverages', 'loss', 'appCoverages']) {
    let t = raw[key]; if (!t) continue;
    t = copy(t);
    const id = key === 'appCoverages' ? 'app_no' : 'policy_no';
    const emptyRows = []; for (let i = 0; i < t.length; i++) if (!t.cols[id][i] || !t.cols.coverage_code[i]) emptyRows.push(i);
    if (emptyRows.length) { add(key, 'error', `${id} 또는 coverage_code가 비어 있는 행`, emptyRows.length, ex(t, emptyRows), '행 제외'); const bad = new Set(emptyRows); t = filterRows(t, i => !bad.has(i)); }
    if (key === 'loss') {
      const neg = [], zero = [], negPaid = []; let negCnt = 0;
      for (let i = 0; i < t.length; i++) {
        const p = t.cols.risk_premium[i];
        if (p != null && p < 0) neg.push(i); else if (p === 0) zero.push(i);
        if (t.cols.claim_paid[i] != null && t.cols.claim_paid[i] < 0) negPaid.push(i);
      }
      if (neg.length) { add(key, 'error', '위험보험료가 음수인 행', neg.length, ex(t, neg, 'risk_premium'), '행 제외'); const bad = new Set(neg); t = filterRows(t, i => !bad.has(i)); }
      if (zero.length) add(key, 'warn', '위험보험료가 0인 행 (손해만 있고 보험료가 없음)', zero.length, ex(t, zero), '유지 (확인 필요)');
      if (negPaid.length) add(key, 'info', '지급보험금이 음수인 행 (환입 등)', negPaid.length, ex(t, negPaid, 'claim_paid'), '유지');
      if (t.cols.claim_cnt) {
        t.cols.claim_cnt = t.cols.claim_cnt.map(v => { if (v != null && v < 0) { negCnt++; return 0; } return v; });
        if (negCnt) add(key, 'fix', '사고 건수가 음수인 값', negCnt, [], '0으로 변경');
      }
      const seen = new Set(); let dup = 0; for (let i = 0; i < t.length; i++) { const k = t.cols.policy_no[i] + '|' + t.cols.coverage_code[i]; if (seen.has(k)) dup++; else seen.add(k); }
      if (dup) add(key, 'info', '같은 증권·담보가 여러 행 (기간별 실적 등)', dup, [], '증권 단위로 합산');
    }
    if (t.cols.sum_insured) {
      const bad = []; for (let i = 0; i < t.length; i++) { const v = t.cols.sum_insured[i]; if (v == null || v <= 0) bad.push(i); }
      if (bad.length) add(key, 'warn', '가입금액이 없거나 0 이하', bad.length, ex(t, bad, 'sum_insured'), key === 'appCoverages' ? '권고 가입금액 계산 불가' : '가입금액대 분석에서 제외');
    }
    if (t.cols.coverage_name) {
      const broken = []; for (let i = 0; i < t.length; i++) if (String(t.cols.coverage_name[i]).includes('\uFFFD')) broken.push(i);
      if (broken.length) add(key, 'error', '담보명 글자가 깨졌습니다 (인코딩 확인 필요)', broken.length, ex(t, broken, 'coverage_name'), '부담보 판정 불가');
    }
    out[key] = t;
  }

  if (raw.credit) {
    let t = copy(raw.credit); const bad = [];
    t.cols.credit_decile = t.cols.credit_decile.map((v, i) => { if (v == null) return null; if (!Number.isInteger(v) || v < 1 || v > CONFIG.DECILES) { bad.push(i); return null; } return v; });
    if (bad.length) add('credit', 'error', `credit_decile이 1~${CONFIG.DECILES} 정수가 아님`, bad.length, ex(raw.credit, bad, 'credit_decile'), '해당 고객 분석 제외');
    const seen = new Set(); let dup = 0; for (const c of t.cols.customer_id) { if (seen.has(c)) dup++; else seen.add(c); }
    if (dup) add('credit', 'warn', 'customer_id 중복 (기준일이 여러 개일 수 있음)', dup, [], '마지막 행 사용');
    out.credit = t;
  }
  if (raw.delinq) {
    const t = copy(raw.delinq); let neg = 0, empty = 0;
    for (const c of [...CONFIG.DELINQ.DEFAULT_COLS, ...CONFIG.DELINQ.DELINQ_COLS]) {
      t.cols[c] = t.cols[c].map(v => { if (v == null) { empty++; return 0; } if (v < 0) { neg++; return 0; } return v; });
    }
    if (neg) add('delinq', 'fix', '연체·채무불이행 건수가 음수인 값', neg, [], '0으로 변경');
    if (empty) add('delinq', 'info', '연체·채무불이행 건수가 비어 있는 값', empty, [], '0(없음)으로 처리');
    out.delinq = t;
  }

  // 3) 파일 간 점검
  const names = new Map();
  for (const key of ['coverages', 'appCoverages']) {
    const t = out[key]; if (!t) continue;
    for (let i = 0; i < t.length; i++) { const c = t.cols.coverage_code[i]; let set = names.get(c); if (!set) names.set(c, set = new Set()); set.add(t.cols.coverage_name[i]); }
  }
  const conflict = [...names].filter(([, s]) => s.size > 1);
  if (conflict.length) add(null, 'warn', '같은 담보코드에 담보명이 여러 개', conflict.length, conflict.map(([c, s]) => `${c}: ${[...s].join(' / ')}`), '담보명 기준으로 분석');
  if (out.appCoverages) {
    const present = new Set(out.appCoverages.cols.coverage_name), excl = CONFIG.EXCLUSION_COVERAGES;
    const missing = excl.filter(n => !present.has(n));
    if (missing.length === excl.length) add('appCoverages', 'warn', '부담보 대상 담보명이 청약 데이터에 하나도 없습니다', excl.length, excl, 'CONFIG.EXCLUSION_COVERAGES를 실제 담보명으로 수정 필요');
    else if (missing.length) add('appCoverages', 'info', '부담보 대상 중 청약 데이터에 없는 담보명', missing.length, missing, '해당 담보는 부담보 미적용');
  }
  const pairs = [['contracts', 'loss', 'policy_no', '손해실적'], ['contracts', 'credit', 'customer_id', '신용정보'], ['contracts', 'delinq', 'customer_id', '연체정보'],
    ['apps', 'credit', 'customer_id', '신용정보'], ['apps', 'appCoverages', 'app_no', '청약 담보']];
  for (const [a, b, k, label] of pairs) {
    if (!out[a] || !out[b]) continue;
    const keys = new Set(out[b].cols[k]); let miss = 0; const exRows = [];
    for (let i = 0; i < out[a].length; i++) if (!keys.has(out[a].cols[k][i])) { miss++; if (exRows.length < 3) exRows.push(`${k}=${out[a].cols[k][i]}`); }
    const rate = out[a].length ? 1 - miss / out[a].length : 1;
    if (miss) add(a, rate < CONFIG.REVIEW.MIN_MATCH_RATE ? 'warn' : 'info', `${label}와 매칭되지 않는 행 (매칭률 ${(rate * 100).toFixed(1)}%)`, miss, exRows, '분석에서 제외');
  }
  const order = { error: 0, warn: 1, fix: 2, info: 3 };
  issues.sort((x, y) => order[x.level] - order[y.level]);
  const summary = { error: 0, warn: 0, fix: 0, info: 0 }; for (const i of issues) summary[i.level]++;
  return { files: { ...raw, ...out }, issues, summary };
}

function scaleDiagnostics(ds) {
  const items = [], G = CONFIG.DELINQ.LABELS;
  const byDec = new Array(CONFIG.DECILES).fill(0), byGrade = [0, 0, 0];
  if (ds.hasClaimCnt) {
    for (let k = 0; k < ds.cov.length; k++) { const p = ds.policies[ds.cov.pol[k]]; byDec[p.decile - 1] += ds.cov.claims[k]; if (p.grade != null) byGrade[p.grade] += ds.cov.claims[k]; }
    const minD = Math.min(...byDec);
    items.push({ level: minD < CONFIG.REVIEW.MIN_DECILE_CLAIMS ? 'warn' : 'info', message: `10분위별 사고 건수 최소 ${minD.toLocaleString()}건 (기준 ${CONFIG.REVIEW.MIN_DECILE_CLAIMS}건)`,
      examples: byDec.map((c, i) => `${i + 1}분위 ${c.toLocaleString()}`), action: minD < CONFIG.REVIEW.MIN_DECILE_CLAIMS ? '부분 신뢰: 95% 구간으로 판단, 대형사고 상한·분위 합치기 검토' : '' });
    if (ds.hasDelinq) {
      const minG = Math.min(...byGrade);
      items.push({ level: minG < CONFIG.REVIEW.MIN_GRADE_CLAIMS ? 'warn' : 'info', message: `연체등급별 사고 건수 최소 ${minG.toLocaleString()}건 (기준 ${CONFIG.REVIEW.MIN_GRADE_CLAIMS}건)`,
        examples: byGrade.map((c, g) => `${G[g]} ${c.toLocaleString()}`), action: minG < CONFIG.REVIEW.MIN_GRADE_CLAIMS ? '부분 신뢰: 연체 효과는 95% 구간으로 판단, 등급 합치기 검토' : '' });
    }
  } else items.push({ level: 'warn', message: 'claim_cnt(사고 건수) 컬럼이 없어 규모 진단·빈도 기준을 쓸 수 없습니다', examples: [], action: '' });
  return items.map(x => ({ file: '전체 (규모 진단)', count: null, ...x }));
}

도구 설명
- 신용 10분위(1 우량 ~ 10 불량)와 연체등급(0 정상, 1 연체, 2 채무불이행)별 건강보험 손해율을 분석해 심사 스코어를 만들고, 청약에 권고 인수조건과 담보별 권고 가입금액을 보여준다.
- 손해율 = (Σ보험금 + Σ준비금변동) ÷ Σ위험보험료
- 10분위 상대도 = 분위 손해율 ÷ 전체 손해율
- 연체 보정 상대도 = 등급 Σ실제손해 ÷ 등급 Σ(위험보험료 × 전체손해율 × 10분위 상대도). 신용평점에 이미 반영된 몫을 빼서 이중반영을 막는다.
- 종합상대도 A = 10분위 상대도, B = 10분위 상대도 × 연체 보정 상대도. 스코어 = 100 ÷ 종합상대도 (높을수록 우량)
- 권고(B): 1.0 미만 표준인수, 1.0~1.3 한도축소(전 담보 70%), 1.3~1.6 부담보(질병수술·재해수술·질병입원·재해입원) + 나머지 담보 50%, 1.6 이상 심사역검토
- 분리도 = 종합상대도 순 10등분에서 최악 10% 손해율 ÷ 최우량 10% 손해율. 같은 데이터로 정렬·채점한 값은 과적합으로 부풀려지므로 교차 검증 값을 주 지표로 말한다
- 95% 구간이 서로 겹치거나 1을 포함하면 차이나 효과를 단정하지 않는다
- 손해 기준(원손해, 대형사고 상한, 빈도 기준)은 고액·희귀 사고에 따른 흔들림을 줄이는 선택지다
- 데이터는 증권×담보 단위 가상 데이터 v4(증권 10만, 담보 14개, 청약 1,000건)다. 실손은 없다.

- 신용정보만을 이유로 한 인수 거절은 허용되지 않을 수 있고, 실제 적용 전 법무·준법 검토가 필요하며, 최종 판단은 심사역이 한다는 원칙을 지킨다.
- 개인을 특정하거나 차별을 정당화하는 답은 하지 않는다.
- 간결하게 답한다. 필요하면 짧은 목록을 쓴다.`;

function bad(res, status, message) {
  res.statusCode = status;
  res.setHeader('Content-Type', 'application/json; charset=utf-8');
  res.end(JSON.stringify({ error: message }));
}

async function readJson(req) {
  if (req.body && typeof req.body === 'object') return req.body;
  if (typeof req.body === 'string') return JSON.parse(req.body);
  let raw = '';
  for await (const chunk of req) {
    raw += chunk;
    if (raw.length > 500000) throw new Error('too large');
  }
  return JSON.parse(raw || '{}');
}

// 같은 사이트에서 온 요청만 받는다 (다른 사이트가 키를 대신 쓰는 것을 막는 최소한의 장치)
function sameOrigin(req) {
  const origin = req.headers.origin;
  if (!origin) return true;
  try { return new URL(origin).host === req.headers.host; } catch { return false; }
}

function sanitizeMessages(list) {
  if (!Array.isArray(list)) return null;
  const out = list
    .filter(m => m && (m.role === 'user' || m.role === 'assistant') && typeof m.content === 'string')
    .slice(-LIMITS.messages)
    .map(m => ({ role: m.role, content: m.content.slice(0, LIMITS.messageChars) }));
  return out.length && out[out.length - 1].role === 'user' ? out : null;
}

module.exports = async function handler(req, res) {
  if (req.method !== 'POST') return bad(res, 405, 'POST만 지원합니다');
  if (!sameOrigin(req)) return bad(res, 403, '허용되지 않은 출처입니다');
  const key = process.env.OPENROUTER_API_KEY;
  if (!key) return bad(res, 503, 'Vercel 환경변수 OPENROUTER_API_KEY가 설정되지 않았습니다');

  let body;
  try { body = await readJson(req); } catch { return bad(res, 400, '요청 형식이 올바르지 않습니다'); }
  const messages = sanitizeMessages(body.messages);
  if (!messages) return bad(res, 400, '질문이 비어 있습니다');
  const context = typeof body.context === 'string' ? body.context.slice(0, LIMITS.contextChars) : '';

  const models = process.env.OPENROUTER_MODEL
    ? process.env.OPENROUTER_MODEL.split(',').map(s => s.trim()).filter(Boolean)
    : DEFAULT_MODELS;

  let upstream;
  try {
    upstream = await fetch('https://openrouter.ai/api/v1/chat/completions', {
      method: 'POST',
      headers: {
        Authorization: `Bearer ${key}`,
        'Content-Type': 'application/json',
        'HTTP-Referer': `https://${req.headers.host || 'score-poc'}`,
        'X-Title': 'Score PoC AI Helper',
      },
      body: JSON.stringify({
        model: models[0],
        ...(models.length > 1 ? { models } : {}),
        stream: true,
        max_tokens: LIMITS.maxTokens,
        temperature: 0.3,
        reasoning: { exclude: true },
        messages: [
          { role: 'system', content: SYSTEM_PROMPT },
          { role: 'system', content: `[현재 화면 요약]\n${context || '(아직 분석을 실행하지 않음)'}` },
          ...messages,
        ],
      }),
    });
  } catch {
    return bad(res, 502, 'AI 서버에 연결하지 못했습니다');
  }

  if (!upstream.ok || !upstream.body) {
    let detail = '';
    try { detail = (await upstream.json())?.error?.message || ''; } catch { /* 본문 없음 */ }
    const msg = upstream.status === 429 ? '무료 모델 사용량 한도에 걸렸습니다. 잠시 후 다시 시도해 주세요'
      : upstream.status === 401 ? 'OpenRouter API 키가 유효하지 않습니다'
      : `AI 응답 오류 (${upstream.status})${detail ? ': ' + detail.slice(0, 200) : ''}`;
    return bad(res, 502, msg);
  }

  res.statusCode = 200;
  res.setHeader('Content-Type', 'text/event-stream; charset=utf-8');
  res.setHeader('Cache-Control', 'no-cache, no-transform');
  const reader = upstream.body.getReader();
  try {
    for (;;) {
      const { done, value } = await reader.read();
      if (done) break;
      res.write(Buffer.from(value));
    }
  } catch {
    res.write('data: {"error":"응답 스트림이 끊겼습니다"}\n\n');
  }
  res.end();
};
